    thresholds_array = np.array(thresholds_array)

    print("\n" + "─"*58)
    print(f"F1 macro  seuil=0.5    : {f1_05:.4f}")
    print(f"F1 macro  seuil optimal : {f1_opt:.4f}")
    print(f"F1 micro  seuil optimal : {f1_micro:.4f}")

    return thresholds_array
